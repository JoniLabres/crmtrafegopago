Meta Conversions API (CAPI)
Meta CAPI — payload Python (backend)
Execute no servidor no momento da conversão Meta CAPI
Envio server-side para a CAPI do Meta. Contorna iOS/bloqueadores. Use em paralelo ao pixel client-side com event_id igual para deduplicação automática.
import hashlib, time, uuid, requests, os

def sha256(value: str) -> str:
    return hashlib.sha256(value.strip().lower().encode()).hexdigest()

def send_capi_lead(
    email: str,
    phone: str = "",
    first_name: str = "",
    utm_campaign: str = "",
    utm_source: str = "",
    client_ip: str = "",
    client_ua: str = "",
    event_source_url: str = "",
):
    pixel_id    = "{{ ap.meta or 'SEU_PIXEL_ID' }}"
    access_token = os.getenv("META_ACCESS_TOKEN")
    event_id    = str(uuid.uuid4())

    user_data = {"em": sha256(email)}
    if phone:      user_data["ph"] = sha256(phone)
    if first_name: user_data["fn"] = sha256(first_name)
    if client_ip:  user_data["client_ip_address"] = client_ip
    if client_ua:  user_data["client_user_agent"]  = client_ua

    payload = {
        "data": [{
            "event_name": "Lead",
            "event_time": int(time.time()),
            "event_id":   event_id,    # igual ao event_id do pixel client-side
            "event_source_url": event_source_url,
            "action_source": "website",
            "user_data": user_data,
            "custom_data": {
                "utm_campaign": utm_campaign,
                "utm_source":   utm_source,
                "currency": "BRL",
                "value": 0,
            },
        }],
        "test_event_code": "",  # remova em produção
    }

    r = requests.post(
        f"https://graph.facebook.com/v19.0/{pixel_id}/events",
        params={"access_token": access_token},
        json=payload,
        timeout=10,
    )
    r.raise_for_status()
    return {"event_id": event_id, "result": r.json()}
