var _utms = (window.IXC && window.IXC.getUtmLast()) || {};

// Meta Pixel — Lead
if(typeof fbq !== 'undefined') {
  fbq('track', 'Lead', {
    utm_campaign: _utms.utm_campaign || '',
    utm_source:   _utms.utm_source   || '',
    currency: 'BRL', value: 0,
  });
}

// TikTok — SubmitForm
if(typeof ttq !== 'undefined') ttq.track('SubmitForm');

// LinkedIn — Conversão
if(typeof lintrk !== 'undefined') lintrk('track', { conversion_id: '' });

// dataLayer (para GA4 e qualquer outro via GTM)
window.dataLayer = window.dataLayer || [];
window.dataLayer.push({
  event: 'generate_lead',
  utm_source:   _utms.utm_source   || '',
  utm_medium:   _utms.utm_medium   || '',
  utm_campaign: _utms.utm_campaign || '',
  form_id: 'seu-formulario',
